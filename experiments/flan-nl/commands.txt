# Arm F-nl of the FLAN-decay proof on mbert-corpus (v4-32, us-central2-b, 4 workers), from /Users/yeb/Developer/yhavinga/hammond.
# Protocol: protocol.md, copied to $R before any run of F-nl. Code: branch flan-nl of github.com/yhavinga/mbert-jax at
# COMMIT_NL (translation and mix; scripts in $B), the trainer at F's commit F_COMMIT (start_flan_nl.sh checks it out in
# ~/mbert-flan). Everything of the translation lives in /dev/shm/flan_nl on the workers (the boot disks have ~9 GB free);
# a reboot empties it and setup.sh rebuilds it.
# One thing at a time on the TPU: translate.sh refuses while anything holds /dev/accel*; stop a hung one with
# pkill -9 -f translate.py. Do not touch mbert-mix.
COMMIT_NL=<the commit of this file on flan-nl>
F_COMMIT=bd6ddca909168bd4fb1dc9cd1cc748bb861a53c5
R=gs://t5-dutch-english-usc2/mbert/flan-nl
B=$R/scripts
S=gs://t5-dutch-english-usc2/mbert/culturax-nl-230m-chunk1024/ckpt_keep/653000
PY='$HOME/mbert-dumb/.venv/bin/python'
SSH='gcloud compute tpus tpu-vm ssh mbert-corpus --zone us-central2-b --project t5-dutch-english --quiet'
CODE=/dev/shm/flan_nl/code
VENV=/dev/shm/flan_nl/venv/bin/python

# 0. From the Mac, in a checkout of flan-nl at COMMIT_NL: scripts and protocol into the bucket
gsutil -m cp experiments/flan-nl/*.sh experiments/flan-nl/*.py experiments/flan-nl/protocol.md experiments/flan-nl/commands.txt $B/ && gsutil cp experiments/flan-nl/protocol.md $R/protocol.md

# 1. Environment on every worker, and on worker 0 the conversion of TranslateGemma 27B into $R/translategemma-27b-maxtext
# (~2 minutes of install; on worker 0 1 minute of download and 6 of conversion). Log ~/logs/flan_nl_setup.log, ends in EXIT.
# The HF token is there from F's setup (flan-decay/commands.txt, step 1).
$SSH --worker=all --command "mkdir -p $CODE ~/logs && gsutil -q cp '$B/*' $CODE/ && W=\$(hostname | sed 's/.*-w-//') && ARG= && { [ \$W = 0 ] && ARG=convert || true; } && (setsid nohup bash -c \"bash $CODE/setup.sh \$ARG > ~/logs/flan_nl_setup.log 2>&1; echo EXIT \\\$? >> ~/logs/flan_nl_setup.log\" > /dev/null 2>&1 < /dev/null &)" < /dev/null
$SSH --worker=all --command 'tail -n 2 ~/logs/flan_nl_setup.log' < /dev/null
# 12B, the model of the first run (stopped, protocol.md "Data of F-nl"): its conversion on worker 3 into $R/translategemma-12b-maxtext,
# and its tokenizer on every worker (the same tokenizer and template as 27B's)
$SSH --worker=3 --command "(MODEL=12b setsid nohup bash -c 'bash $CODE/setup.sh convert > ~/logs/flan_nl_setup12b.log 2>&1; echo EXIT \$? >> ~/logs/flan_nl_setup12b.log' > /dev/null 2>&1 < /dev/null &)" < /dev/null
$SSH --worker=all --command "MODEL=12b bash $CODE/setup.sh" < /dev/null
# The safetensors are not needed after the conversions (52 and 24 GB of RAM):
$SSH --worker=0 --command 'rm -f /dev/shm/flan_nl/hf-27b/*.safetensors' < /dev/null
$SSH --worker=3 --command 'rm -f /dev/shm/flan_nl/hf-12b/*.safetensors' < /dev/null

# 2. The FLAN rows of F's mix of every host (seconds): $R/en/w<N>.parquet
$SSH --worker=all --command "W=\$(hostname | sed 's/.*-w-//') && $VENV $CODE/flan_nl.py extract --rank \$W --list ~/flan_decay_files.txt --out /dev/shm/flan_nl/en-w\$W.parquet && gsutil -q cp /dev/shm/flan_nl/en-w\$W.parquet $R/en/w\$W.parquet" < /dev/null

# 3. The plan on worker 0 (~2 minutes): task filters, pieces, distinct segments by worker -> $R/plan/
$SSH --worker=0 --command "mkdir -p /dev/shm/flan_nl/plan /dev/shm/flan_nl/en && for w in 0 1 2 3; do gsutil -q cp $R/en/w\$w.parquet /dev/shm/flan_nl/en/w\$w.parquet; done && TOKENIZERS_PARALLELISM=false $VENV -u $CODE/flan_nl.py plan --english /dev/shm/flan_nl/en/w{0,1,2,3}.parquet --tokenizer /dev/shm/flan_nl/tokenizer-27b --out /dev/shm/flan_nl/plan --processes 96 > ~/logs/flan_nl_plan.log 2>&1; echo EXIT \$? >> ~/logs/flan_nl_plan.log; gsutil -m -q cp /dev/shm/flan_nl/plan/* $R/plan/" < /dev/null
gsutil cat $R/plan/plan.json

# 4. Pilot: 300 random examples to translate (seed 0), translated, assembled. 27B first (on worker 0, $R/pilot/out/27b,
# $R/pilot/nl), then 12B, the model of the run (on worker 2, $R/pilot/out/12b, $R/pilot/nl12b). Throughput per class:
# the SHARD lines of the logs; the first shards of the run are a random sample of each class (translate.py).
$SSH --worker=0 --command "mkdir -p /dev/shm/flan_nl/pilot && $VENV $CODE/flan_nl.py sample --plan /dev/shm/flan_nl/plan --out /dev/shm/flan_nl/pilot --n 300 --seed 0 && gsutil -m -q cp /dev/shm/flan_nl/pilot/* $R/pilot/" < /dev/null
$SSH --worker=2 --command "SEGMENTS=$R/pilot/segments-w0.parquet OUT=$R/pilot/out/12b/w0 NAME=flan_nl_pilot12 bash $CODE/translate.sh" < /dev/null
$SSH --worker=2 --command "grep -a '^SHARD\|^DONE\|^EXIT\|Error' ~/logs/flan_nl_pilot12.log" < /dev/null
$SSH --worker=2 --command "mkdir -p /dev/shm/flan_nl/pilot/nl12b /dev/shm/flan_nl/pilot/out12b && gsutil -q cp '$R/plan/*' /dev/shm/flan_nl/plan/ && gsutil -m -q cp $R/en/w0.parquet $R/en/w1.parquet $R/en/w2.parquet $R/en/w3.parquet /dev/shm/flan_nl/en/ && gsutil -m -q cp $R/pilot/examples.parquet $R/pilot/segments-w0.parquet /dev/shm/flan_nl/pilot/ && gsutil -m -q rsync -r $R/pilot/out/12b /dev/shm/flan_nl/pilot/out12b && $VENV $CODE/flan_nl.py assemble --english /dev/shm/flan_nl/en/w{0,1,2,3}.parquet --plan /dev/shm/flan_nl/pilot --translations /dev/shm/flan_nl/pilot/out12b --out /dev/shm/flan_nl/pilot/nl12b --processes 8 && gsutil -m -q cp -r /dev/shm/flan_nl/pilot/nl12b $R/pilot/" < /dev/null

# 5. The template strings (the 30 most frequent labels and opening sentences of F's FLAN rows, with their 27B and 12B
# renderings) -> $R/templates/top30.{md,json} = experiments/flan-nl/templates_top30.{md,json}. Yeb's phrasing table
# templates_nl.json (31 keys: the 30 and Tail:) is in the branch.
$SSH --worker=0 --command "mkdir -p /dev/shm/flan_nl/tpl/27b /dev/shm/flan_nl/tpl/12b /dev/shm/flan_nl/templates && gsutil -m -q rsync -r $R/pilot/out/27b /dev/shm/flan_nl/tpl/27b/pilot && gsutil -m -q rsync -r $R/out/27b /dev/shm/flan_nl/tpl/27b/out && gsutil -m -q rsync -r $R/pilot/out/12b /dev/shm/flan_nl/tpl/12b/pilot && gsutil -m -q rsync -r $R/translations/12b /dev/shm/flan_nl/tpl/12b/run && $VENV $CODE/flan_nl.py templates --english /dev/shm/flan_nl/en/w{0,1,2,3}.parquet --plan /dev/shm/flan_nl/plan --out27b /dev/shm/flan_nl/tpl/27b --out12b /dev/shm/flan_nl/tpl/12b --out /dev/shm/flan_nl/templates && gsutil -q cp /dev/shm/flan_nl/templates/top30.* $R/templates/" < /dev/null

# 6. The plan of the 27B run, with the template strings protected (~3 minutes) -> $R/plan2/
$SSH --worker=0 --command "mkdir -p /dev/shm/flan_nl/plan2 && TOKENIZERS_PARALLELISM=false $VENV -u $CODE/flan_nl.py plan --english /dev/shm/flan_nl/en/w{0,1,2,3}.parquet --tokenizer /dev/shm/flan_nl/tokenizer-27b --templates $CODE/templates_top30.json $CODE/templates_extra.json $CODE/templates_nl.json --out /dev/shm/flan_nl/plan2 --processes 96 > ~/logs/flan_nl_plan2.log 2>&1; echo EXIT \$? >> ~/logs/flan_nl_plan2.log; gsutil -m -q cp /dev/shm/flan_nl/plan2/* $R/plan2/" < /dev/null

# 7. The full translation with 27B, every worker its quarter, class after class (tiny, small, medium, long, xlong;
# shards in $R/translations/27b/w<N>/<class>/; ~12 hours estimated). The same command resumes after a crash or a
# reboot (after step 1 again): stored shards are skipped. Stop it with pkill -9 -f '[t]ranslate.py'.
# (The 12B run of 8 Nov was this command with MODEL=12b SEGMENTS=$R/plan/segments-w<N>.parquet and four classes;
# stopped at about a third, its shards in $R/translations/12b/.)
$SSH --worker=all --command "bash $CODE/translate.sh" < /dev/null
# Progress and the end (each log ends in EXIT 0):
$SSH --worker=all --command "grep -a '^SHARD' ~/logs/flan_nl_translate.log | tail -n 1; grep -a '^DONE\|^EXIT\|Error' ~/logs/flan_nl_translate.log" < /dev/null
for w in 0 1 2 3; do echo w$w $(for c in tiny small medium long xlong; do gsutil ls $R/translations/27b/w$w/$c/ 2>/dev/null | wc -l; done); done   # of 57, 31, 12, 6 and 8 shards

# 8. Assembly on worker 0 (~15 minutes): the Dutch rows of every host, the template variants, what was dropped why -> $R/nl/
$SSH --worker=0 --command "mkdir -p /dev/shm/flan_nl/out27b /dev/shm/flan_nl/nl && gsutil -m -q rsync -r $R/translations/27b /dev/shm/flan_nl/out27b && $VENV -u $CODE/flan_nl.py assemble --english /dev/shm/flan_nl/en/w{0,1,2,3}.parquet --plan /dev/shm/flan_nl/plan2 --translations /dev/shm/flan_nl/out27b --templates-nl $CODE/templates_nl.json --out /dev/shm/flan_nl/nl --processes 96 > ~/logs/flan_nl_assemble.log 2>&1; echo EXIT \$? >> ~/logs/flan_nl_assemble.log; gsutil -m -q cp /dev/shm/flan_nl/nl/* $R/nl/" < /dev/null
gsutil cat $R/nl/report.json

# 9. The Dutch mix of every host (CPU, the trainer's venv; ~20 minutes): ~/flan_decay_nl/mix-<build>.parquet,
# ~/flan_decay_nl_files.txt, the manifest in $R/mix/w<N>.json
$SSH --worker=all --command "(PY=$PY setsid nohup bash -c 'bash $CODE/build_mix.sh > ~/logs/flan_nl_mix.log 2>&1; echo EXIT \$? >> ~/logs/flan_nl_mix.log' > /dev/null 2>&1 < /dev/null &)" < /dev/null
$SSH --worker=all --command 'tail -n 3 ~/logs/flan_nl_mix.log' < /dev/null
for w in 0 1 2 3; do gsutil cat $R/mix/w$w.json | python3 -c "import json, sys; m = json.load(sys.stdin); f = m['flan']; print('w$w', m['rows'], f['f_rows'], f['rows'], round(m['flan_token_share'], 4), round(m['f_flan_token_share'], 4))"; done

# 11. FLAN-nl on Hugging Face (Yeb's consent of 8 Nov: public as yhavinga/flan-nl, every row with its source row in
# Open-Orca/FLAN), worker 0, CPU. publish.py and what it imports (flan_nl.py, scripts/build_decay_mix.py of this branch)
# are in $R/publish/, on the VM in /dev/shm/flan_nl/publish/ (not in $CODE, not in $B). HANDOVER.md, "Publish".
# 11a. link (done 8 Nov 20:21-20:29 UTC, ~8 minutes): F's 1,357,022 FLAN rows -> their file and row in Open-Orca/FLAN at
# F's revision, read from the shards F read (each downloaded to /dev/shm/flan_nl/oo-flan and deleted after) -> $R/links/.
# Log ~/logs/flan_nl_link.log, ends in EXIT 0; then the copy to $R/links/.
$SSH --worker=0 --command "mkdir -p ~/logs /dev/shm/flan_nl/publish/fmix /dev/shm/flan_nl/links /dev/shm/flan_nl/oo-flan && gsutil -q cp '$R/publish/*.py' /dev/shm/flan_nl/publish/ && gsutil -q cp 'gs://t5-dutch-english-usc2/mbert/flan-decay/mix/w*.json' /dev/shm/flan_nl/publish/fmix/ && (setsid nohup bash -c 'HF_HUB_DISABLE_PROGRESS_BARS=1 nice -n 19 $VENV -u /dev/shm/flan_nl/publish/publish.py link --english /dev/shm/flan_nl/en --manifests /dev/shm/flan_nl/publish/fmix --out /dev/shm/flan_nl/links --download-dir /dev/shm/flan_nl/oo-flan > ~/logs/flan_nl_link.log 2>&1; echo EXIT \$? >> ~/logs/flan_nl_link.log' > /dev/null 2>&1 < /dev/null &)" < /dev/null
$SSH --worker=0 --command "grep -a 'LINKED\|done:\|EXIT\|Error' ~/logs/flan_nl_link.log" < /dev/null
$SSH --worker=0 --command "gsutil -m -q cp /dev/shm/flan_nl/links/w0.parquet /dev/shm/flan_nl/links/w1.parquet /dev/shm/flan_nl/links/w2.parquet /dev/shm/flan_nl/links/w3.parquet /dev/shm/flan_nl/links/report.json $R/links/ && gsutil -q cp ~/logs/flan_nl_link.log $R/logs/flan_nl_link.log" < /dev/null
# 11b. build, after step 8 ends in EXIT 0 (~3 minutes, CPU, 32 processes under nice): /dev/shm/flan_nl/hf3/
# {data/train-*.parquet,README.md,build_report.json} -> $R/hf3/. The checks of this publication (decode_loop,
# english_left; publish.py, HANDOVER.md "Publish") read the English extracts --english; build_report.json has the rows
# they left out, per check by task and by id (not uploaded). It refuses a non-empty --out: a rerun goes to a new
# directory. After a reboot the links come from $R/links/, the extracts from $R/en/. Not to be uploaded: the first
# build (9 Nov 03:17 UTC by the VMs' clock, without these checks; /dev/shm/flan_nl/hf, $R/hf/) and hf2 (04:24, the
# same rows as hf3, its card says the rows left out were all in the training data; /dev/shm/flan_nl/hf2, $R/hf2/).
$SSH --worker=0 --command "mkdir -p ~/logs /dev/shm/flan_nl/publish /dev/shm/flan_nl/links /dev/shm/flan_nl/en && gsutil -q cp '$R/publish/*.py' /dev/shm/flan_nl/publish/ && { [ -s /dev/shm/flan_nl/links/report.json ] || gsutil -m -q cp '$R/links/*' /dev/shm/flan_nl/links/; } && { [ -s /dev/shm/flan_nl/en/w3.parquet ] || gsutil -m -q cp '$R/en/*' /dev/shm/flan_nl/en/; } && (setsid nohup bash -c 'nice -n 19 $VENV -u /dev/shm/flan_nl/publish/publish.py build --nl /dev/shm/flan_nl/nl --links /dev/shm/flan_nl/links --plan /dev/shm/flan_nl/plan2 --english /dev/shm/flan_nl/en --processes 32 --out /dev/shm/flan_nl/hf3 > ~/logs/flan_nl_publish_build3.log 2>&1; echo EXIT \$? >> ~/logs/flan_nl_publish_build3.log' > /dev/null 2>&1 < /dev/null &)" < /dev/null
$SSH --worker=0 --command "tail -n 3 ~/logs/flan_nl_publish_build3.log" < /dev/null   # BUILD {...} and EXIT 0
$SSH --worker=0 --command "gsutil -m -q cp -r /dev/shm/flan_nl/hf3 $R/ && gsutil -q cp ~/logs/flan_nl_publish_build3.log $R/logs/flan_nl_publish_build3.log" < /dev/null
gsutil cat $R/hf3/README.md   # the dataset card: read it before 11c
# 11c. upload: a public dataset repo (created if absent) with README.md and data/ of hf3 (not build_report.json); prints the URL
$SSH --worker=0 --command "HF_HUB_DISABLE_PROGRESS_BARS=1 $VENV -u /dev/shm/flan_nl/publish/publish.py upload --folder /dev/shm/flan_nl/hf3 --repo yhavinga/flan-nl --public" < /dev/null
